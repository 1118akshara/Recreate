import android.os.Bundle
import android.widget.TextView
import androidx.appcompat.app.AppCompatActivity
import com.btech.recreate.R
import com.btech.recreate.network.BackendApi
import org.json.JSONArray

class AIReuseDetailActivity : AppCompatActivity() {

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_ai_reuse_detail)

        val idea = intent.getStringExtra("idea_title")

        findViewById<TextView>(R.id.txtIdeaTitle).text = idea

        if (idea == null) return

        BackendApi.generateSteps(idea) { result ->

            runOnUiThread {

                if (result == null) {
                    findViewById<TextView>(R.id.txtSteps).text =
                        "Failed to load steps"
                    return@runOnUiThread
                }

                try {
                    val array = JSONArray(result)

                    val builder = StringBuilder()

                    for (i in 0 until array.length()) {
                        val step = array.getJSONObject(i)

                        builder.append("Step ${step.getInt("step_number")}\n")
                        builder.append(step.getString("instruction"))
                        builder.append("\n\n")
                    }

                    findViewById<TextView>(R.id.txtSteps).text =
                        builder.toString()

                } catch (e: Exception) {
                    findViewById<TextView>(R.id.txtSteps).text =
                        "Invalid step response"
                }
            }
        }
    }
}

package com.btech.recreate.activities

import android.content.Intent
import android.os.Bundle
import android.widget.ArrayAdapter
import android.widget.ListView
import androidx.appcompat.app.AppCompatActivity
import com.btech.recreate.R
import com.btech.recreate.network.BackendApi
import org.json.JSONObject
import androidx.appcompat.app.AlertDialog

class AIReuseSuggestionsActivity : AppCompatActivity() {

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_ai_reuse_suggestions)

        title = "Reuse Suggestions"

        val analysisJson = intent.getStringExtra("analysis_json")

        if (analysisJson == null) {
            showError("No analysis data found")
            return
        }

        BackendApi.generateAIReuseIdeas(analysisJson) { result ->

            runOnUiThread {

                if (result == null) {
                    showError("Failed to get AI suggestions")
                    return@runOnUiThread
                }

                try {
                    val json = JSONObject(result)

                    val easy = json.getJSONArray("easy")
                    val moderate = json.getJSONArray("moderate")
                    val hard = json.getJSONArray("very_hard")

                    setupList(R.id.listEasy, easy)
                    setupList(R.id.listModerate, moderate)
                    setupList(R.id.listHard, hard)

                } catch (e: Exception) {
                    showError("Invalid AI response")
                }
            }
        }
    }

    private fun setupList(listId: Int, array: org.json.JSONArray) {

        val listView = findViewById<ListView>(listId)

        val titles = mutableListOf<String>()

        for (i in 0 until array.length()) {
            titles.add(array.getJSONObject(i).getString("title"))
        }

        val adapter = ArrayAdapter(
            this,
            android.R.layout.simple_list_item_1,
            titles
        )

        listView.adapter = adapter

        listView.setOnItemClickListener { _, _, position, _ ->

            val intent = Intent(
                this,
                AIReuseDetailActivity::class.java
            )

            intent.putExtra("idea_title", titles[position])
            startActivity(intent)
        }
    }

    private fun showError(message: String) {
        AlertDialog.Builder(this)
            .setMessage(message)
            .setPositiveButton("OK", null)
            .show()
    }
}

        val txtObjects = findViewById<TextView>(R.id.txtObjects)
        val txtReuse = findViewById<TextView>(R.id.txtReuse)

        if (jsonString.isNullOrBlank()) {

        }

        try {
            val json = JSONObject(jsonString)

            val objectsArray = json.optJSONArray("objects")
            val combinedReuseArray = json.optJSONArray("combined_reuse_suggestions")

            if (objectsArray == null || objectsArray.length() == 0) {
                txtObjects.text = "No objects detected."
                txtReuse.text = ""
                return
            }

            val objectsText = StringBuilder()
            objectsText.append("Detected Objects\n\n")

                val obj = objectsArray.getJSONObject(i)

                val material = obj.optString("material", "Unknown")

                val height = obj
                    .optJSONObject("dimensions")
                    ?.optDouble("height_cm", 0.0)
                    ?: 0.0

                objectsText.append("â€¢ Object ${i + 1}\n")
                objectsText.append("  Type: $name\n")
                objectsText.append("  Material: $material\n")
                objectsText.append("  Height: ${"%.2f".format(height)} cm\n\n")

            txtObjects.text = objectsText.toString()

            val reuseText = StringBuilder()

            if (combinedReuseArray != null && combinedReuseArray.length() > 0) {
                reuseText.append("Combined Reuse Suggestions\n\n")
                for (i in 0 until combinedReuseArray.length()) {
                    reuseText.append("â€¢ ${combinedReuseArray.getString(i)}\n")
                }
            } else {
                reuseText.append("No combined reuse suggestions available.")
            }

            txtReuse.text = reuseText.toString()

        } catch (e: Exception) {
            e.printStackTrace()
            txtObjects.text = "Failed to parse analysis result."
            txtReuse.text = ""
        }

        // NAVIGATION TO REUSE SUGGESTIONS PAGE
        btnAIReuse.setOnClickListener {

            val intent = Intent(
                this@AnalysisResultActivity2,
                AIReuseSuggestionsActivity::class.java

            )
            intent.putExtra("analysis_json", jsonString)
            startActivity(intent)
        }
    }

package com.btech.recreate.analysis

import com.btech.recreate.data.ScanRepository
import org.json.JSONArray
import org.json.JSONObject

object AnalysisPayloadBuilder {

    fun buildPayload(): JSONObject {
        val root = JSONObject()
        val objectsArray = JSONArray()

        ScanRepository.scannedObjects.forEachIndexed { index, obj ->
            val objJson = JSONObject()
            objJson.put("object_id", "object_${index + 1}")

            val imagesArray = JSONArray()

            obj.captures.values.forEach { frame ->
                val imageJson = JSONObject()
                imageJson.put("view", frame.viewType.name)
                imageJson.put("image_path", frame.imagePath)
                imageJson.put("pose", JSONArray(frame.poseMatrix.toList()))
                imagesArray.put(imageJson)
            }

            objJson.put("images", imagesArray)
            objectsArray.put(objJson)
        }

        root.put("objects", objectsArray)
        return root
    }
}

package com.btech.recreate.network

import android.util.Log
import okhttp3.*
import okhttp3.MediaType.Companion.toMediaType
import okhttp3.RequestBody.Companion.asRequestBody
import org.json.JSONObject
import java.io.File
import java.io.IOException

object BackendApi {

    private val client = OkHttpClient()

    /* ==========================================================
       1ï¸âƒ£ ANALYZE SESSION (YOLO + EfficientNet)
     ========================================================== */

    fun uploadMultipleObjects(
        objectImageMap: Map<String, Map<String, File>>,
        onResult: (String?) -> Unit
    ) {

        val multipart = MultipartBody.Builder()
            .setType(MultipartBody.FORM)

        for ((objectId, views) in objectImageMap) {
            for ((viewType, file) in views) {

                multipart.addFormDataPart("object_ids", objectId)
                multipart.addFormDataPart("view_types", viewType)

                val body = file
                    .asRequestBody("image/jpeg".toMediaType())

                multipart.addFormDataPart(
                    "files",
                    file.name,
                    body
                )
            }
        }

        val request = Request.Builder()
            .url("${ServerConfig.BASE_URL}/analyze-session")
            .post(multipart.build())
            .build()

        client.newCall(request).enqueue(object : Callback {

            override fun onFailure(call: Call, e: IOException) {
                onResult(null)
            }

            override fun onResponse(call: Call, response: Response) {
                onResult(response.body?.string())
            }
        })
    }

    /* ==========================================================
       2ï¸âƒ£ GENERATE AI REUSE IDEAS
     ========================================================== */

    fun generateAIReuseIdeas(
        analysisJson: String,
        onResult: (String?) -> Unit
    ) {

        val mediaType = "application/json".toMediaType()

        val body = RequestBody.create(
            mediaType,
            analysisJson
        )

        val request = Request.Builder()
            .url("${ServerConfig.BASE_URL}/generate-reuse-ai")
            .post(body)
            .build()

        client.newCall(request).enqueue(object : Callback {

            override fun onFailure(call: Call, e: IOException) {
                onResult(null)
            }

            override fun onResponse(call: Call, response: Response) {
                onResult(response.body?.string())
            }
        })
    }

    /* ==========================================================
       3ï¸âƒ£ GENERATE STEPS
     ========================================================== */

    fun generateSteps(
        ideaTitle: String,
        onResult: (String?) -> Unit
    ) {

        val json = JSONObject()
        json.put("idea", ideaTitle)

        val body = RequestBody.create(
            "application/json".toMediaType(),
            json.toString()
        )

        val request = Request.Builder()
            .url("${ServerConfig.BASE_URL}/generate-steps")
            .post(body)
            .build()

        client.newCall(request).enqueue(object : Callback {

            override fun onFailure(call: Call, e: IOException) {
                onResult(null)
            }

            override fun onResponse(call: Call, response: Response) {
                onResult(response.body?.string())
            }
        })
    }
}

    )

def predict_material(image_paths):

    votes = []

    for image_path in image_paths:

        if not os.path.exists(image_path):
            continue

        image = Image.open(image_path).convert("RGB")

        # ðŸ”¥ Use full image for stability
        cropped = image

        tensor = transform(cropped).unsqueeze(0).to(device)

        with torch.no_grad():
            outputs = material_model(tensor)
            probs = torch.softmax(outputs, dim=1)
            class_idx = torch.argmax(probs, dim=1).item()
            confidence = probs[0][class_idx].item()

        # ðŸ”¥ Stronger confidence threshold
        if confidence >= 0.60:
            predicted_material = MATERIAL_CLASSES[class_idx]
            votes.append((predicted_material, confidence))

    # No valid predictions
    if not votes:
        return {
            "material": "unknown",
            "confidence": 0.0
        }

    # Confidence-weighted voting
    score_map = {}
    for material, conf in votes:
        score_map[material] = score_map.get(material, 0) + conf

    final_material = max(score_map, key=score_map.get)
    final_confidence = score_map[final_material] / len(votes)

    # ðŸ”¥ Filter unwanted materials
    if final_material not in ALLOWED_MATERIALS:
        final_material = "unknown"

    return {
        "material": final_material,
        "confidence": round(final_confidence, 2)
    }

from ultralytics import YOLO
from collections import Counter
import os

MODEL_PATH = os.path.join(os.path.dirname(__file__), "yolov8l.pt")
model = YOLO(MODEL_PATH)

CONFIDENCE_THRESHOLD = 0.60

ALLOWED_OBJECTS = [
    "bottle", "cup", "bowl", "wine glass",
    "chair", "couch", "bed", "dining table",
    "tv", "laptop", "mouse", "remote", "keyboard",
    "cell phone", "microwave", "oven", "toaster",
    "refrigerator", "hair drier",
    "book", "clock", "vase", "scissors",
    "backpack", "suitcase", "umbrella", "handbag",
    "fork", "knife", "spoon"
]


def detect_object(image_paths):

    labels = []

    for image_path in image_paths:
        results = model(image_path, verbose=False)

        for r in results:
            if not r.boxes:
                continue

            for box in r.boxes:

                conf = float(box.conf[0])
                cls_id = int(box.cls[0])
                label = model.names.get(cls_id)

                if (
                    conf >= CONFIDENCE_THRESHOLD
                    and label in ALLOWED_OBJECTS
                ):
                    labels.append(label)

    if not labels:
        return "unknown"

    return Counter(labels).most_common(1)[0][0]

import torch
from torchvision import datasets, models, transforms
import torch.nn as nn
import torch.optim as optim
import os

print("Starting training script...")

DATASET_DIR = "material_dataset"
EPOCHS = 5
BATCH_SIZE = 32
MODEL_OUT = "efficientnet_material.pth"

device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
print("Using device:", device)

# Transforms
transform = transforms.Compose([
    transforms.Resize((224, 224)),
    transforms.ToTensor(),
    transforms.Normalize(
        mean=[0.485, 0.456, 0.406],
        std=[0.229, 0.224, 0.225]
    )
])

# Dataset paths
train_path = os.path.join(DATASET_DIR, "train")
val_path = os.path.join(DATASET_DIR, "val")

print("Loading dataset from:", train_path)

train_ds = datasets.ImageFolder(train_path, transform=transform)
val_ds = datasets.ImageFolder(val_path, transform=transform)

print("Classes found:", train_ds.classes)
print("Training images:", len(train_ds))
print("Validation images:", len(val_ds))

train_loader = torch.utils.data.DataLoader(
    train_ds, batch_size=BATCH_SIZE, shuffle=True
)

# Model
model = models.efficientnet_b0(pretrained=True)
model.classifier[1] = nn.Linear(
    model.classifier[1].in_features,
    len(train_ds.classes)
)
model.to(device)

criterion = nn.CrossEntropyLoss()
optimizer = optim.Adam(model.parameters(), lr=0.0003)

# Training loop
for epoch in range(EPOCHS):
    model.train()
    total_loss = 0

    for images, labels in train_loader:
        images, labels = images.to(device), labels.to(device)

        optimizer.zero_grad()
        outputs = model(images)
        loss = criterion(outputs, labels)
        loss.backward()
        optimizer.step()

        total_loss += loss.item()

    print(f"Epoch {epoch+1}/{EPOCHS} - Loss: {total_loss:.4f}")

# Save model
torch.save(model.state_dict(), MODEL_OUT)
print("Training complete. Model saved as:", MODEL_OUT)
