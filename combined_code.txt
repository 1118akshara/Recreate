        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_ai_reuse_detail)

        val idea = intent.getStringExtra("idea_title")

        findViewById<TextView>(R.id.txtIdeaTitle).text = idea

        if (idea == null) return

        BackendApi.generateSteps(idea) { result ->

            runOnUiThread {

                if (result == null) {
                    findViewById<TextView>(R.id.txtSteps).text =
                        "Failed to load steps"
                    return@runOnUiThread
                }

                try {
                    val array = JSONArray(result)

                    val builder = StringBuilder()

                    for (i in 0 until array.length()) {
                        val step = array.getJSONObject(i)

                        builder.append("Step ${step.getInt("step_number")}\n")
                        builder.append(step.getString("instruction"))
                        builder.append("\n\n")
                    }

                    findViewById<TextView>(R.id.txtSteps).text =
                        builder.toString()

                } catch (e: Exception) {
                    findViewById<TextView>(R.id.txtSteps).text =
                        "Invalid step response"
                }
            }
        }
    }
}

class AIReuseSuggestionsActivity : AppCompatActivity() {

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_ai_reuse_suggestions)

        title = "Reuse Suggestions"

        val analysisJson = intent.getStringExtra("analysis_json")

        if (analysisJson == null) {
            showError("No analysis data found")
            return
        }

        BackendApi.generateAIReuseIdeas(analysisJson) { result ->

            runOnUiThread {

                if (result == null) {
                    showError("Failed to get AI suggestions")
                    return@runOnUiThread
                }

                try {
                    val json = JSONObject(result)

                    val easy = json.getJSONArray("easy")
                    val moderate = json.getJSONArray("moderate")
                    val hard = json.getJSONArray("very_hard")

                    setupList(R.id.listEasy, easy)
                    setupList(R.id.listModerate, moderate)
                    setupList(R.id.listHard, hard)

                } catch (e: Exception) {
                    showError("Invalid AI response")
                }
            }
        }
    }

    private fun setupList(listId: Int, array: org.json.JSONArray) {

        val listView = findViewById<ListView>(listId)

        val titles = mutableListOf<String>()

        for (i in 0 until array.length()) {
            titles.add(array.getJSONObject(i).getString("title"))
        }

        val adapter = ArrayAdapter(
            this,
            android.R.layout.simple_list_item_1,
            titles
        )

        listView.adapter = adapter

        listView.setOnItemClickListener { _, _, position, _ ->

            val intent = Intent(
                this,
                AIReuseDetailActivity::class.java
            )

            intent.putExtra("idea_title", titles[position])
            startActivity(intent)
        }
    }

    private fun showError(message: String) {
        AlertDialog.Builder(this)
            .setMessage(message)
            .setPositiveButton("OK", null)
            .show()
    }
}

package com.btech.recreate.activities

import android.content.Intent
import android.graphics.Bitmap
import android.os.Bundle
import android.os.Environment
import android.widget.Button
import android.widget.TextView
import android.widget.Toast
import androidx.appcompat.app.AppCompatActivity
import com.btech.recreate.R
import com.btech.recreate.ar.ARImageUtil
import com.btech.recreate.ar.ARRulerManager
import com.btech.recreate.data.ObjectScanSession
import com.btech.recreate.data.ScanRepository
import com.google.ar.core.Plane
import com.google.ar.sceneform.ux.ArFragment
import java.io.File
import java.io.FileOutputStream
import java.util.UUID

class ARScanActivity : AppCompatActivity(),
    ARRulerManager.MeasurementListener {

    private lateinit var arFragment: ArFragment
    private lateinit var txtInstruction: TextView
    private lateinit var txtObjectInfo: TextView
    private lateinit var txtLiveMeasurement: TextView
    private lateinit var btnCapture: Button
    private lateinit var rulerManager: ARRulerManager

    private var measurementStep = 0
    private var objectCount = 1

    private var currentObject =
        ObjectScanSession(UUID.randomUUID().toString())

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_ar_scan)

        objectCount = ScanRepository.scannedObjects.size + 1

        arFragment =
            supportFragmentManager.findFragmentById(R.id.arFragment) as ArFragment

        txtInstruction = findViewById(R.id.txt_instruction)
        txtObjectInfo = findViewById(R.id.txt_object_info)
        txtLiveMeasurement = findViewById(R.id.txt_live_measurement)
        btnCapture = findViewById(R.id.btn_capture_view)

        rulerManager = ARRulerManager(this, arFragment, this)

        // âœ… DO NOT MODIFY SESSION HERE

        arFragment.setOnTapArPlaneListener { hitResult, plane, _ ->

            if (measurementStep < 3 &&
                (plane.type == Plane.Type.HORIZONTAL_UPWARD_FACING ||
                        plane.type == Plane.Type.VERTICAL)
            ) {
                rulerManager.addPoint(hitResult)
            }
        }

        btnCapture.setOnClickListener {
            if (measurementStep < 3) {
                Toast.makeText(
                    this,
                    "Complete all measurements first",
                    Toast.LENGTH_SHORT
                ).show()
            } else {
                captureImage()
            }
        }

        updateUI()
    }

    override fun onMeasured(distanceCm: Float) {

        txtLiveMeasurement.text = "%.2f cm".format(distanceCm)

        when (measurementStep) {
            0 -> currentObject.heightCm = distanceCm
            1 -> currentObject.widthCm = distanceCm
            2 -> currentObject.depthCm = distanceCm
        }

        measurementStep++
        updateUI()
    }

    private fun updateUI() {

        txtObjectInfo.text = "Object $objectCount"

        when (measurementStep) {
            0 -> txtInstruction.text = "Measure Height (tap two points)"
            1 -> txtInstruction.text = "Measure Width (tap two points)"
            2 -> txtInstruction.text = "Measure Depth (tap two points)"
            else -> {
                txtInstruction.text = "All dimensions captured\nTap Capture Image"
                btnCapture.isEnabled = true
            }
        }
    }

    private fun captureImage() {

        ARImageUtil.capture(this, arFragment.arSceneView) { bitmap ->

            if (bitmap == null) {
                Toast.makeText(this, "Capture failed", Toast.LENGTH_SHORT).show()
                return@capture
            }

            val imagePath = saveBitmap(bitmap)
            currentObject.imagePath = imagePath

            ScanRepository.scannedObjects.add(currentObject)

            Toast.makeText(this, "Object $objectCount saved!", Toast.LENGTH_SHORT).show()

            startActivity(Intent(this, ScanSummaryActivity::class.java))
            finish()
        }
    }

    private fun saveBitmap(bitmap: Bitmap): String {

        val dir = File(
            getExternalFilesDir(Environment.DIRECTORY_PICTURES),
            "recreate_scans"
        )
        if (!dir.exists()) dir.mkdirs()

        val file = File(dir, "obj${objectCount}.jpg")

        FileOutputStream(file).use {
            bitmap.compress(Bitmap.CompressFormat.JPEG, 90, it)
        }

        return file.absolutePath
    }
}

package com.btech.recreate.ar

import android.app.Activity
import android.graphics.Bitmap
import android.os.Handler
import android.os.Looper
import android.view.PixelCopy
import com.google.ar.sceneform.ArSceneView

object ARImageUtil {

    fun capture(
        activity: Activity,
        view: ArSceneView,
        callback: (Bitmap?) -> Unit
    ) {
        val bitmap =
            Bitmap.createBitmap(view.width, view.height, Bitmap.Config.ARGB_8888)

        PixelCopy.request(
            view,
            bitmap,
            { result ->
                if (result == PixelCopy.SUCCESS) {
                    callback(bitmap)
                } else {
                    callback(null)
                }
            },
            Handler(Looper.getMainLooper())
        )
    }
}

package com.btech.recreate.ar

import android.animation.ValueAnimator
import android.content.Context
import android.view.animation.LinearInterpolator
import com.google.ar.core.HitResult
import com.google.ar.sceneform.AnchorNode
import com.google.ar.sceneform.Node
import com.google.ar.sceneform.math.Quaternion
import com.google.ar.sceneform.math.Vector3
import com.google.ar.sceneform.rendering.Color
import com.google.ar.sceneform.rendering.MaterialFactory
import com.google.ar.sceneform.rendering.ShapeFactory
import com.google.ar.sceneform.ux.ArFragment

class ARRulerManager(
    private val context: Context,
    private val arFragment: ArFragment,
    private val listener: MeasurementListener
) {

    interface MeasurementListener {
        fun onMeasured(distanceCm: Float)
    }

    private var firstPoint: AnchorNode? = null
    private var secondPoint: AnchorNode? = null
    private var lineNode: Node? = null
    private var animator: ValueAnimator? = null

    fun addPoint(hitResult: HitResult) {

        if (firstPoint != null && secondPoint != null) {
            clear()
        }

        val anchor = hitResult.createAnchor()
        val anchorNode = AnchorNode(anchor)
        anchorNode.setParent(arFragment.arSceneView.scene)

        MaterialFactory.makeOpaqueWithColor(
            context,
            Color(android.graphics.Color.RED)
        ).thenAccept { material ->
            val sphere = ShapeFactory.makeSphere(
                0.015f,
                Vector3.zero(),
                material
            )
            anchorNode.renderable = sphere
        }

        if (firstPoint == null) {
            firstPoint = anchorNode
        } else {
            secondPoint = anchorNode
            drawLine()
        }
    }

    private fun drawLine() {

        val start = firstPoint!!.worldPosition
        val end = secondPoint!!.worldPosition

        val difference = Vector3.subtract(end, start)
        val distanceMeters = difference.length()
        val distanceCm = distanceMeters * 100f

        val midpoint = Vector3.add(start, end).scaled(0.5f)

        MaterialFactory.makeOpaqueWithColor(
            context,
            Color(android.graphics.Color.YELLOW)
        ).thenAccept { material ->

            val cylinder = ShapeFactory.makeCylinder(
                0.004f,
                distanceMeters,
                Vector3.zero(),
                material
            )

            lineNode = Node()
            lineNode!!.setParent(arFragment.arSceneView.scene)
            lineNode!!.renderable = cylinder
            lineNode!!.worldPosition = midpoint

            val direction = difference.normalized()
            val rotation = Quaternion.rotationBetweenVectors(
                Vector3(0f, 1f, 0f),
                direction
            )

            lineNode!!.worldRotation = rotation

            startGlow(material)
        }

        listener.onMeasured(distanceCm)
    }

    private fun startGlow(material: com.google.ar.sceneform.rendering.Material) {

        animator = ValueAnimator.ofFloat(0.5f, 1f)
        animator?.duration = 1000
        animator?.repeatCount = ValueAnimator.INFINITE
        animator?.repeatMode = ValueAnimator.REVERSE
        animator?.interpolator = LinearInterpolator()

        animator?.addUpdateListener {
            val value = it.animatedValue as Float
            material.setFloat3("baseColorTint", value, value, 0f)
        }

        animator?.start()
    }

    fun clear() {
        animator?.cancel()
        animator = null
        firstPoint?.setParent(null)
        secondPoint?.setParent(null)
        lineNode?.setParent(null)
        firstPoint = null
        secondPoint = null
        lineNode = null
    }
}

        planeDiscoveryController.setInstructionView(null)

    }

package com.btech.recreate.network

import okhttp3.*
import okhttp3.MediaType.Companion.toMediaType
import okhttp3.RequestBody.Companion.asRequestBody
import org.json.JSONObject
import java.io.File
import java.io.IOException
import com.btech.recreate.data.ObjectScanSession

object BackendApi {

    private val client = OkHttpClient()

    /* ==========================================================
       1ï¸âƒ£ UPLOAD SESSION (YOLO + EfficientNet)
     ========================================================== */

    fun uploadSession(
        objects: List<ObjectScanSession>,
        onResult: (String?) -> Unit
    ) {

        val multipart = MultipartBody.Builder()
            .setType(MultipartBody.FORM)

        objects.forEachIndexed { index, obj ->

            val file = File(obj.imagePath ?: return@forEachIndexed)
            if (!file.exists()) return@forEachIndexed

            val body =
                file.asRequestBody("image/jpeg".toMediaType())

            // IMPORTANT: must match FastAPI
            multipart.addFormDataPart(
                "files",
                file.name,
                body
            )

            multipart.addFormDataPart(
                "height_cm_$index",
                obj.heightCm?.toString() ?: ""
            )

            multipart.addFormDataPart(
                "width_cm_$index",
                obj.widthCm?.toString() ?: ""
            )

            multipart.addFormDataPart(
                "depth_cm_$index",
                obj.depthCm?.toString() ?: ""
            )
        }

        val request = Request.Builder()
            .url("${ServerConfig.BASE_URL}/analyze-session")
            .post(multipart.build())
            .build()

        client.newCall(request).enqueue(object : Callback {

            override fun onFailure(call: Call, e: IOException) {
                onResult(null)
            }

            override fun onResponse(call: Call, response: Response) {
                onResult(response.body?.string())
            }
        })
    }

    /* ==========================================================
       2ï¸âƒ£ GENERATE AI REUSE IDEAS
     ========================================================== */

    fun generateAIReuseIdeas(
        analysisJson: String,
        onResult: (String?) -> Unit
    ) {

        val body = RequestBody.create(
            "application/json".toMediaType(),
            analysisJson
        )

        val request = Request.Builder()
            .url("${ServerConfig.BASE_URL}/generate-reuse-ai")
            .post(body)
            .build()

        client.newCall(request).enqueue(object : Callback {

            override fun onFailure(call: Call, e: IOException) {
                onResult(null)
            }

            override fun onResponse(call: Call, response: Response) {
                onResult(response.body?.string())
            }
        })
    }

    /* ==========================================================
       3ï¸âƒ£ GENERATE STEPS
     ========================================================== */

    fun generateSteps(
        ideaTitle: String,
        onResult: (String?) -> Unit
    ) {

        val json = JSONObject()
        json.put("idea", ideaTitle)

        val body = RequestBody.create(
            "application/json".toMediaType(),
            json.toString()
        )

        val request = Request.Builder()
            .url("${ServerConfig.BASE_URL}/generate-steps")
            .post(body)
            .build()

        client.newCall(request).enqueue(object : Callback {

            override fun onFailure(call: Call, e: IOException) {
                onResult(null)
            }

            override fun onResponse(call: Call, response: Response) {
                onResult(response.body?.string())
            }
        })
    }
}

import torch
from torchvision import models, transforms
from PIL import Image
import os

# ===============================
# ORIGINAL TRAINED MATERIAL CLASSES (DO NOT MODIFY COUNT)
# ===============================
MATERIAL_CLASSES = [
    "brick",
    "carpet",
    "ceramic",
    "fabric",
    "foliage",
    "food",
    "glass",
    "hair",
    "leather",
    "metal",
    "mirror",
    "other",
    "painted",
    "paper",
    "plastic",
    "polishedstone",
    "skin",
    "sky",
    "stone",
    "tile",
    "wallpaper",
    "water",
    "wood"
]

NUM_CLASSES = len(MATERIAL_CLASSES)

# ===============================
# ALLOWED MATERIALS FOR REUSE SYSTEM
# ===============================
ALLOWED_MATERIALS = [
    "brick",
    "carpet",
    "fabric",
    "foliage",
    "food",
    "glass",
    "hair",
    "leather",
    "metal",
    "mirror",
    "other",
    "painted",
    "paper",
    "plastic",
    "polishedstone",
    "stone",
    "wallpaper",
    "wood"
]

# ===============================
# DEVICE
# ===============================
device = torch.device("cuda" if torch.cuda.is_available() else "cpu")

# ===============================
# LOAD EFFICIENTNET
# ===============================
material_model = models.efficientnet_b0(weights=None)
material_model.classifier[1] = torch.nn.Linear(1280, NUM_CLASSES)

BASE_DIR = os.path.dirname(__file__)
MODEL_PATH = os.path.join(BASE_DIR, "efficientnet_material.pth")

state_dict = torch.load(MODEL_PATH, map_location=device)
material_model.load_state_dict(state_dict)

material_model.to(device)
material_model.eval()

# ===============================
# IMAGE PREPROCESSING
# ===============================
transform = transforms.Compose([
    transforms.Resize((224, 224)),
    transforms.ToTensor(),
    transforms.Normalize(
        mean=[0.485, 0.456, 0.406],
        std=[0.229, 0.224, 0.225]
    )
])

# ===============================
# MATERIAL PREDICTION
# ===============================
def predict_material(image_paths):

    votes = []

    for image_path in image_paths:

        if not os.path.exists(image_path):
            continue

        image = Image.open(image_path).convert("RGB")

        # ðŸ”¥ Use full image for stability
        cropped = image

        tensor = transform(cropped).unsqueeze(0).to(device)

        with torch.no_grad():
            outputs = material_model(tensor)
            probs = torch.softmax(outputs, dim=1)
            class_idx = torch.argmax(probs, dim=1).item()
            confidence = probs[0][class_idx].item()

        # ðŸ”¥ Stronger confidence threshold
        if confidence >= 0.60:
            predicted_material = MATERIAL_CLASSES[class_idx]
            votes.append((predicted_material, confidence))

    # No valid predictions
    if not votes:
        return {
            "material": "unknown",
            "confidence": 0.0
        }

    # Confidence-weighted voting
    score_map = {}
    for material, conf in votes:
        score_map[material] = score_map.get(material, 0) + conf

    final_material = max(score_map, key=score_map.get)
    final_confidence = score_map[final_material] / len(votes)

    # ðŸ”¥ Filter unwanted materials
    if final_material not in ALLOWED_MATERIALS:
        final_material = "unknown"

    return {
        "material": final_material,
        "confidence": round(final_confidence, 2)
    }

from ultralytics import YOLO
from collections import Counter
import os

MODEL_PATH = os.path.join(os.path.dirname(__file__), "yolov8l.pt")
model = YOLO(MODEL_PATH)

CONFIDENCE_THRESHOLD = 0.60

ALLOWED_OBJECTS = [
    "bottle", "cup", "bowl", "wine glass",
    "chair", "couch", "bed", "dining table",
    "tv", "laptop", "mouse", "remote", "keyboard",
    "cell phone", "microwave", "oven", "toaster",
    "refrigerator", "hair drier",
    "book", "clock", "vase", "scissors",
    "backpack", "suitcase", "umbrella", "handbag",
    "fork", "knife", "spoon"
]


def detect_object(image_paths):

    labels = []

    for image_path in image_paths:
        results = model(image_path, verbose=False)

        for r in results:
            if not r.boxes:
                continue

            for box in r.boxes:

                conf = float(box.conf[0])
                cls_id = int(box.cls[0])
                label = model.names.get(cls_id)

                if (
                    conf >= CONFIDENCE_THRESHOLD
                    and label in ALLOWED_OBJECTS
                ):
                    labels.append(label)

    if not labels:
        return "unknown"

    return Counter(labels).most_common(1)[0][0]

import os
import json
from dotenv import load_dotenv
from openai import OpenAI

# Load environment variables
load_dotenv()

api_key = os.getenv("OPENAI_API_KEY")

if not api_key:
    raise ValueError("OPENAI_API_KEY not found. Check your .env file location.")

client = OpenAI(api_key=api_key)


def generate_steps(idea_title: str):
    """
    Generates structured step-by-step instructions
    for a selected reuse idea.
    """

    prompt = f"""
You are an expert DIY instructor.

Create detailed, beginner-friendly step-by-step instructions for:

{idea_title}

For each step include:
- step_number
- instruction
- tools_required
- image_prompt (clear description for generating illustration image)

Rules:
- Keep steps realistic
- Use simple tools
- Make it safe
- 5â€“8 steps maximum

Return JSON ONLY in this format:

[
  {{
    "step_number": 1,
    "instruction": "",
    "tools_required": "",
    "image_prompt": ""
  }}
]
"""

    response = client.chat.completions.create(
        model="gpt-4o-mini",
        messages=[
            {"role": "user", "content": prompt}
        ],
        temperature=0.7
    )

    content = response.choices[0].message.content

    # Ensure valid JSON is returned
    try:
        return json.loads(content)
    except json.JSONDecodeError:
        # If AI slightly misformats JSON, return raw text for debugging
        return {
            "error": "Invalid JSON returned by AI",
            "raw_response": content
        }
